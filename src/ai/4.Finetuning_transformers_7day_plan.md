### 🧩 Part 1: Scaled Dot-Product Attention

```python
import torch
import torch.nn as nn
import torch.nn.functional as F
import math
import numpy as np
import matplotlib.pyplot as plt

def scaled_dot_product_attention(
    Q: torch.Tensor,
    K: torch.Tensor,
    V: torch.Tensor,
    mask: torch.Tensor = None,
    dropout: float = 0.0
) -> tuple[torch.Tensor, torch.Tensor]:
    """
    Scaled Dot-Product Attention (Vaswani et al., 2017)

    Formula: Attention(Q, K, V) = softmax(QK^T / sqrt(d_k)) * V

    Args:
        Q:    (batch, ..., seq_q, d_k)   Queries
        K:    (batch, ..., seq_k, d_k)   Keys
        V:    (batch, ..., seq_k, d_v)   Values
        mask: (batch, ..., seq_q, seq_k) 0 = masked, 1 = attend
        dropout: dropout probability on attention weights

    Returns:
        output:  (batch, ..., seq_q, d_v)  Attended values
        weights: (batch, ..., seq_q, seq_k) Attention weights
    """
    d_k = Q.shape[-1]

    # Step 1: Raw attention scores
    # Q @ K^T: (batch, seq_q, d_k) @ (batch, d_k, seq_k) = (batch, seq_q, seq_k)
    scores = torch.matmul(Q, K.transpose(-2, -1))  # (..., seq_q, seq_k)

    # Step 2: Scale — prevents dot products from growing too large
    # Without scaling: large d_k → huge dot products → peaky softmax → vanishing gradients
    scores = scores / math.sqrt(d_k)

    # Step 3: Apply mask (set masked positions to -inf so softmax → 0)
    if mask is not None:
        scores = scores.masked_fill(mask == 0, float('-inf'))

    # Step 4: Softmax → probability distribution over keys for each query
    weights = F.softmax(scores, dim=-1)             # (..., seq_q, seq_k)

    # Handle NaN from all-masked rows (softmax of all -inf)
    weights = torch.nan_to_num(weights, nan=0.0)

    # Optional dropout on attention weights
    if dropout > 0.0 and torch.is_grad_enabled():
        weights = F.dropout(weights, p=dropout)

    # Step 5: Weighted sum of values
    output = torch.matmul(weights, V)               # (..., seq_q, d_v)

    return output, weights


# ── Shape verification ────────────────────────────────────────────────
def verify_attention_shapes():
    batch, seq, d_k, d_v = 2, 10, 64, 64

    Q = torch.randn(batch, seq, d_k)
    K = torch.randn(batch, seq, d_k)
    V = torch.randn(batch, seq, d_v)

    out, w = scaled_dot_product_attention(Q, K, V)

    print("Shape verification:")
    print(f"  Q:       {Q.shape}")        # (2, 10, 64)
    print(f"  K:       {K.shape}")        # (2, 10, 64)
    print(f"  V:       {V.shape}")        # (2, 10, 64)
    print(f"  Output:  {out.shape}")      # (2, 10, 64)
    print(f"  Weights: {w.shape}")        # (2, 10, 10)
    print(f"  Weights sum to 1: {w.sum(dim=-1).allclose(torch.ones(batch, seq))}")

verify_attention_shapes()
```

### 🧩 Part 2: Masking — Padding and Causal

```python
def create_padding_mask(token_ids: torch.Tensor, pad_idx: int = 0) -> torch.Tensor:
    """
    Padding mask: prevents attending to [PAD] tokens.
    Returns 1 for real tokens, 0 for padding.

    Args:
        token_ids: (batch, seq_len)
    Returns:
        mask: (batch, 1, 1, seq_len) — broadcastable for MHA
    """
    # 1 where token is NOT padding, 0 where it IS padding
    mask = (token_ids != pad_idx).float()          # (batch, seq_len)
    return mask.unsqueeze(1).unsqueeze(2)          # (batch, 1, 1, seq_len)


def create_causal_mask(seq_len: int, device: str = 'cpu') -> torch.Tensor:
    """
    Causal mask: prevents attending to future tokens.
    Returns lower-triangular matrix of ones.

    Returns:
        mask: (1, 1, seq_len, seq_len) — broadcastable for MHA
    """
    # Lower triangular: position i can see positions 0..i
    mask = torch.tril(torch.ones(seq_len, seq_len, device=device))
    return mask.unsqueeze(0).unsqueeze(0)          # (1, 1, seq_len, seq_len)


def create_combined_mask(token_ids: torch.Tensor, pad_idx: int = 0) -> torch.Tensor:
    """
    Combines causal mask + padding mask for decoder self-attention.
    """
    seq_len = token_ids.shape[1]
    pad_mask    = create_padding_mask(token_ids, pad_idx)       # (batch, 1, 1, seq_len)
    causal_mask = create_causal_mask(seq_len, token_ids.device) # (1, 1, seq_len, seq_len)
    return pad_mask * causal_mask                               # (batch, 1, seq_len, seq_len)
```

### 🧩 Part 3: Single-Head Self-Attention Module

```python
class SingleHeadSelfAttention(nn.Module):
    """
    Self-attention with learned Q, K, V projections.
    d_model → project → d_k → attend → d_k → project back → d_model
    """

    def __init__(self, d_model: int, d_k: int = None, dropout: float = 0.1):
        super().__init__()
        self.d_k = d_k or d_model

        # Learned projection matrices
        self.W_Q = nn.Linear(d_model, self.d_k, bias=False)
        self.W_K = nn.Linear(d_model, self.d_k, bias=False)
        self.W_V = nn.Linear(d_model, self.d_k, bias=False)
        self.W_O = nn.Linear(self.d_k, d_model, bias=False)  # Output projection

        self.dropout = dropout
        self._init_weights()

    def _init_weights(self):
        """Xavier initialization for stable training."""
        for module in [self.W_Q, self.W_K, self.W_V, self.W_O]:
            nn.init.xavier_uniform_(module.weight)

    def forward(
        self,
        x: torch.Tensor,
        mask: torch.Tensor = None
    ) -> tuple[torch.Tensor, torch.Tensor]:
        """
        Args:
            x:    (batch, seq_len, d_model)
            mask: (batch, 1, seq_len, seq_len)

        Returns:
            output:  (batch, seq_len, d_model)
            weights: (batch, seq_len, seq_len)
        """
        # Project to Q, K, V
        Q = self.W_Q(x)   # (batch, seq_len, d_k)
        K = self.W_K(x)   # (batch, seq_len, d_k)
        V = self.W_V(x)   # (batch, seq_len, d_k)

        # Attention
        attn_out, weights = scaled_dot_product_attention(
            Q, K, V, mask=mask,
            dropout=self.dropout if self.training else 0.0
        )

        # Output projection
        output = self.W_O(attn_out)   # (batch, seq_len, d_model)

        return output, weights
```

### 🧩 Multi-Head Attention (Full Implementation)

```python
class MultiHeadAttention(nn.Module):
    """
    Multi-Head Attention as used in BERT/GPT-style Transformers.

    Architecture:
    Input → W_Q,W_K,W_V → Split into H heads → Parallel attention
    → Concatenate → W_O → Output
    """

    def __init__(self, d_model: int, num_heads: int, dropout: float = 0.1):
        super().__init__()
        assert d_model % num_heads == 0, \
            f"d_model ({d_model}) must be divisible by num_heads ({num_heads})"

        self.d_model   = d_model
        self.num_heads = num_heads
        self.d_k       = d_model // num_heads  # Dimension per head

        # Single matrix for all heads (efficient: one matmul for all H heads)
        self.W_Q = nn.Linear(d_model, d_model, bias=True)
        self.W_K = nn.Linear(d_model, d_model, bias=True)
        self.W_V = nn.Linear(d_model, d_model, bias=True)
        self.W_O = nn.Linear(d_model, d_model, bias=True)

        self.attn_dropout = nn.Dropout(dropout)
        self._init_weights()

    def _init_weights(self):
        for module in [self.W_Q, self.W_K, self.W_V, self.W_O]:
            nn.init.xavier_uniform_(module.weight)
            nn.init.zeros_(module.bias)

    def split_heads(self, x: torch.Tensor) -> torch.Tensor:
        """
        Split last dimension into (num_heads, d_k).
        (batch, seq, d_model) → (batch, num_heads, seq, d_k)
        """
        batch, seq, _ = x.shape
        x = x.view(batch, seq, self.num_heads, self.d_k)
        return x.transpose(1, 2)   # (batch, num_heads, seq, d_k)

    def merge_heads(self, x: torch.Tensor) -> torch.Tensor:
        """
        Merge heads back.
        (batch, num_heads, seq, d_k) → (batch, seq, d_model)
        """
        batch, _, seq, _ = x.shape
        x = x.transpose(1, 2)                         # (batch, seq, num_heads, d_k)
        return x.contiguous().view(batch, seq, self.d_model)

    def forward(
        self,
        query:  torch.Tensor,
        key:    torch.Tensor,
        value:  torch.Tensor,
        mask:   torch.Tensor = None
    ) -> tuple[torch.Tensor, torch.Tensor]:
        """
        For self-attention:      query = key = value = x
        For cross-attention:     query = decoder state, key = value = encoder output

        Args:
            query:  (batch, seq_q, d_model)
            key:    (batch, seq_k, d_model)
            value:  (batch, seq_k, d_model)
            mask:   (batch, 1, seq_q, seq_k)  or  (batch, 1, 1, seq_k)

        Returns:
            output:  (batch, seq_q, d_model)
            weights: (batch, num_heads, seq_q, seq_k)
        """
        # ── 1. Linear projections ─────────────────────────────────────
        Q = self.W_Q(query)   # (batch, seq_q, d_model)
        K = self.W_K(key)     # (batch, seq_k, d_model)
        V = self.W_V(value)   # (batch, seq_k, d_model)

        # ── 2. Split into H heads ──────────────────────────────────────
        Q = self.split_heads(Q)   # (batch, H, seq_q, d_k)
        K = self.split_heads(K)   # (batch, H, seq_k, d_k)
        V = self.split_heads(V)   # (batch, H, seq_k, d_k)

        # ── 3. Scaled dot-product attention (all heads in parallel) ────
        # mask needs an extra dim for heads: (batch, 1, seq_q, seq_k)
        attn_out, weights = scaled_dot_product_attention(Q, K, V, mask=mask)
        # attn_out: (batch, H, seq_q, d_k)
        # weights:  (batch, H, seq_q, seq_k)

        # Apply dropout to attention weights
        weights = self.attn_dropout(weights)

        # ── 4. Merge heads ────────────────────────────────────────────
        attn_out = self.merge_heads(attn_out)   # (batch, seq_q, d_model)

        # ── 5. Output projection ──────────────────────────────────────
        output = self.W_O(attn_out)             # (batch, seq_q, d_model)

        return output, weights


# ── Feed-Forward Network ──────────────────────────────────────────────
class FeedForwardNetwork(nn.Module):
    """
    Position-wise FFN: FFN(x) = GELU(xW₁ + b₁)W₂ + b₂
    Expands d_model → d_ff → d_model (typically d_ff = 4 × d_model)
    """

    def __init__(self, d_model: int, d_ff: int = None, dropout: float = 0.1):
        super().__init__()
        d_ff = d_ff or 4 * d_model

        self.linear1 = nn.Linear(d_model, d_ff)
        self.linear2 = nn.Linear(d_ff, d_model)
        self.dropout  = nn.Dropout(dropout)
        self.activation = nn.GELU()   # BERT uses GELU; original Transformer uses ReLU

        nn.init.xavier_uniform_(self.linear1.weight)
        nn.init.xavier_uniform_(self.linear2.weight)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return self.linear2(self.dropout(self.activation(self.linear1(x))))


# ── Transformer Encoder Block ─────────────────────────────────────────
class TransformerEncoderBlock(nn.Module):
    """
    Single Transformer Encoder block — matches DistilBERT's architecture.

    Structure (Pre-LN variant, more stable):
    x → LayerNorm → MHA → Dropout → x (residual)
      → LayerNorm → FFN → Dropout → x (residual)
    """

    def __init__(self, d_model: int, num_heads: int, d_ff: int = None,
                 dropout: float = 0.1):
        super().__init__()
        self.self_attn  = MultiHeadAttention(d_model, num_heads, dropout)
        self.ffn        = FeedForwardNetwork(d_model, d_ff, dropout)
        self.norm1      = nn.LayerNorm(d_model, eps=1e-12)  # eps matches BERT
        self.norm2      = nn.LayerNorm(d_model, eps=1e-12)
        self.dropout    = nn.Dropout(dropout)

    def forward(
        self,
        x:    torch.Tensor,
        mask: torch.Tensor = None
    ) -> tuple[torch.Tensor, torch.Tensor]:
        """
        Args:
            x:    (batch, seq_len, d_model)
            mask: (batch, 1, 1, seq_len) — padding mask

        Returns:
            x:       updated hidden states
            weights: attention weights for visualization
        """
        # Sublayer 1: Self-Attention with Pre-LayerNorm + Residual
        x_norm = self.norm1(x)
        attn_out, weights = self.self_attn(x_norm, x_norm, x_norm, mask)
        x = x + self.dropout(attn_out)

        # Sublayer 2: Feed-Forward with Pre-LayerNorm + Residual
        x_norm = self.norm2(x)
        ffn_out = self.ffn(x_norm)
        x = x + self.dropout(ffn_out)

        return x, weights


# ── Positional Encoding ────────────────────────────────────────────────
class LearnedPositionalEmbedding(nn.Module):
    """
    Learned positional embeddings (BERT/DistilBERT style).
    Each position 0..max_pos gets its own trainable embedding vector.
    """

    def __init__(self, max_position: int, d_model: int, dropout: float = 0.1):
        super().__init__()
        self.pos_embedding = nn.Embedding(max_position, d_model)
        self.dropout        = nn.Dropout(dropout)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """x: (batch, seq_len, d_model)"""
        seq_len   = x.shape[1]
        positions = torch.arange(seq_len, device=x.device).unsqueeze(0)
        return self.dropout(x + self.pos_embedding(positions))


# ── Complete Mini-BERT ─────────────────────────────────────────────────
class MiniBERT(nn.Module):
    """
    A minimal BERT-like encoder, architecturally similar to DistilBERT.

    DistilBERT specs (for reference):
      vocab_size:  30,522
      d_model:     768
      num_heads:   12
      num_layers:  6   (half of BERT's 12)
      d_ff:        3072
      max_pos:     512
    """

    def __init__(
        self,
        vocab_size:  int = 30522,
        d_model:     int = 256,     # Small for demo
        num_heads:   int = 4,
        num_layers:  int = 3,
        d_ff:        int = 1024,
        max_position: int = 512,
        dropout:     float = 0.1,
        num_classes: int = 2,       # Classification head
        pad_idx:     int = 0
    ):
        super().__init__()
        self.pad_idx = pad_idx

        # Token + Position Embeddings
        self.token_embedding = nn.Embedding(vocab_size, d_model, padding_idx=pad_idx)
        self.pos_embedding   = LearnedPositionalEmbedding(max_position, d_model, dropout)
        self.embedding_norm  = nn.LayerNorm(d_model, eps=1e-12)
        self.embedding_dropout = nn.Dropout(dropout)

        # Transformer Encoder Layers
        self.layers = nn.ModuleList([
            TransformerEncoderBlock(d_model, num_heads, d_ff, dropout)
            for _ in range(num_layers)
        ])

        # Classification Head (uses [CLS] token = position 0)
        self.classifier = nn.Sequential(
            nn.Linear(d_model, d_model),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(d_model, num_classes)
        )

        self._init_weights()

    def _init_weights(self):
        nn.init.normal_(self.token_embedding.weight, std=0.02)
        for module in self.modules():
            if isinstance(module, nn.Linear):
                nn.init.normal_(module.weight, std=0.02)
                if module.bias is not None:
                    nn.init.zeros_(module.bias)

    def forward(
        self,
        input_ids:      torch.Tensor,
        attention_mask: torch.Tensor = None
    ) -> dict:
        """
        Args:
            input_ids:      (batch, seq_len) — token IDs
            attention_mask: (batch, seq_len) — 1=real, 0=padding

        Returns:
            dict with logits, hidden_states, attention_weights
        """
        # Create attention mask (batch, 1, 1, seq_len)
        if attention_mask is not None:
            mask = attention_mask.unsqueeze(1).unsqueeze(2).float()
        else:
            mask = None

        # Embeddings
        x = self.token_embedding(input_ids)     # (batch, seq, d_model)
        x = self.pos_embedding(x)
        x = self.embedding_norm(x)
        x = self.embedding_dropout(x)

        # Transformer layers
        all_weights = []
        for layer in self.layers:
            x, w = layer(x, mask)
            all_weights.append(w)

        # Classification: use [CLS] token (position 0)
        cls_output = x[:, 0, :]                 # (batch, d_model)
        logits     = self.classifier(cls_output) # (batch, num_classes)

        return {
            "logits":             logits,
            "last_hidden_state":  x,
            "attention_weights":  all_weights
        }


# ── Full test ────────────────────────────────────────────────────────────

print(f"Total parameters: {total_params:,}")
```

### 📖 DistilBERT Architecture Overview

```python
"""
BERT-base vs. DistilBERT comparison:

┌─────────────────┬──────────────┬──────────────┐
│ Property        │ BERT-base    │ DistilBERT   │
├─────────────────┼──────────────┼──────────────┤
│ Layers          │ 12           │ 6 (50% less) │
```

### 🔄 Tokenization Pipeline

```python
from transformers import AutoTokenizer

class TokenizationPipeline:
    """
    Production-grade tokenization with all edge cases handled.
    """

    def __init__(
        self,
        model_name:  str = 'distilbert-base-uncased',
        max_length:  int = 512,
        label2id:    dict = None
    ):
        self.tokenizer  = AutoTokenizer.from_pretrained(model_name)
        self.max_length = max_length
        self.label2id   = label2id or {}
        self.id2label   = {v: k for k, v in self.label2id.items()}

    def tokenize_batch(self, batch: dict) -> dict:
        """
        Tokenization function for HuggingFace .map()
        Handles:
          - Truncation of long sequences
          - Padding (done dynamically by DataCollator)
          - Special tokens ([CLS], [SEP])
          - Multiple text columns (for NLI: premise + hypothesis)
        """
        # Single-text tasks (classification, NER)
        if 'text' in batch:
            encoded = self.tokenizer(
                batch['text'],
                truncation=True,
                max_length=self.max_length,
                padding=False,  # Dynamic padding in DataCollator — more efficient
            )
        # Paired-text tasks (NLI, QA, similarity)
        elif 'premise' in batch and 'hypothesis' in batch:
            encoded = self.tokenizer(
                batch['premise'],
                batch['hypothesis'],
                truncation=True,
                max_length=self.max_length,
                padding=False,
            )
        else:
            raise ValueError(f"Unknown batch format. Keys: {list(batch.keys())}")

        # Include labels
        if 'label' in batch:
            encoded['labels'] = batch['label']

        return encoded

    def prepare_dataset(self, dataset: DatasetDict) -> DatasetDict:
        """Apply tokenization to entire dataset."""
        tokenized = dataset.map(
            self.tokenize_batch,
            batched=True,
            batch_size=1000,
            remove_columns=['text'],    # Remove original text column
            desc="Tokenizing"
        )
        tokenized.set_format('torch')
        return tokenized


# ── DataLoaders with dynamic padding ─────────────────────────────────
from transformers import DataCollatorWithPadding

def create_dataloaders(
    tokenized_dataset: DatasetDict,
    tokenizer,
    batch_size: int = 32
) -> dict[str, DataLoader]:
    """
    DataCollatorWithPadding pads each BATCH to the longest sequence in that batch.
    This is more efficient than padding everything to max_length.
    Example: batch max_len=45 → pad to 45, not 512.
    """
    collator = DataCollatorWithPadding(tokenizer=tokenizer, return_tensors='pt')

    loaders = {}
    for split in tokenized_dataset:
        shuffle = (split == 'train')
        loaders[split] = DataLoader(
            tokenized_dataset[split],
            batch_size=batch_size,
            shuffle=shuffle,
            collate_fn=collator,
            num_workers=4,
            pin_memory=True    # Faster GPU transfer
        )
        print(f"{split:12}: {len(tokenized_dataset[split]):5} samples, "
              f"{len(loaders[split]):3} batches")

    return loaders


# ── Full demo ────────────────────────────────────────────────────────
```

### 🔧 Method 2: Manual Training Loop (Full Control)

```python
import torch
from torch.optim import AdamW
from transformers import get_linear_schedule_with_warmup
from torch.cuda.amp import autocast, GradScaler

def train_distilbert_manual(
    model, train_loader, val_loader,
    num_epochs: int = 3,
    lr: float = 2e-5,
    warmup_steps: int = 100
):
    device   = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    model    = model.to(device)

    # Optimizer — separate LR for embeddings vs. task head
    no_decay = ["bias", "LayerNorm.weight"]
    optimizer_groups = [
        {"params": [p for n, p in model.named_parameters()
                    if not any(nd in n for nd in no_decay)],
         "weight_decay": 0.01},
        {"params": [p for n, p in model.named_parameters()
                    if any(nd in n for nd in no_decay)],
         "weight_decay": 0.0}
    ]
    optimizer = AdamW(optimizer_groups, lr=lr)

    total_steps = len(train_loader) * num_epochs
    scheduler   = get_linear_schedule_with_warmup(
        optimizer, num_warmup_steps=warmup_steps,
        num_training_steps=total_steps
    )
    scaler = GradScaler()   # For mixed precision

    history = {"train_loss": [], "val_loss": [], "val_acc": []}
    best_val_acc = 0.0

    for epoch in range(num_epochs):
        # ── Training ────────────────────────────────────────────────────
        model.train()
        train_loss   = 0.0
        train_steps  = 0

        for batch in train_loader:
            input_ids      = batch['input_ids'].to(device)
            attention_mask = batch['attention_mask'].to(device)
            labels         = batch['labels'].to(device)

            with autocast():    # Mixed precision forward
                outputs = model(
                    input_ids=input_ids,
                    attention_mask=attention_mask,
                    labels=labels
                )
                loss = outputs.loss

            # Gradient accumulation (if needed)
            scaler.scale(loss).backward()
            scaler.unscale_(optimizer)
            torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0)   # Gradient clipping
            scaler.step(optimizer)
            scaler.update()
            optimizer.zero_grad()
            scheduler.step()

            train_loss  += loss.item()
            train_steps += 1

        avg_train_loss = train_loss / train_steps

        # ── Validation ──────────────────────────────────────────────────
        model.eval()
        val_loss, correct, total = 0.0, 0, 0

        with torch.no_grad():
            for batch in val_loader:
                input_ids      = batch['input_ids'].to(device)
                attention_mask = batch['attention_mask'].to(device)
                labels         = batch['labels'].to(device)

                with autocast():
                    outputs = model(input_ids=input_ids,
                                    attention_mask=attention_mask,
                                    labels=labels)

                val_loss += outputs.loss.item()
                preds     = outputs.logits.argmax(dim=-1)
                correct  += (preds == labels).sum().item()
                total    += labels.shape[0]

        avg_val_loss = val_loss / len(val_loader)
        val_acc      = correct / total

        history["train_loss"].append(avg_train_loss)
        history["val_loss"].append(avg_val_loss)
        history["val_acc"].append(val_acc)

        print(f"Epoch {epoch+1}/{num_epochs} | "
              f"Train Loss: {avg_train_loss:.4f} | "
              f"Val Loss: {avg_val_loss:.4f} | "
              f"Val Acc: {val_acc:.4f}")

        # Save best model
        if val_acc > best_val_acc:
            best_val_acc = val_acc
            torch.save(model.state_dict(), f'{OUTPUT_DIR}/best_model.pt')
            print(f"  → New best model saved (acc={val_acc:.4f})")

    return history
```

```python
# ── Creating CLM training data ─────────────────────────────────────────
def prepare_clm_dataset(texts: list[str], tokenizer, max_length: int = 256):
    """
    For Causal LM:
      - Concatenate all texts with EOS separator
      - Chunk into fixed-length sequences
      - Labels = input_ids shifted left by 1 (done automatically by DataCollator)

    This "block" approach is more efficient than padding each text separately.
    """
    # Add EOS token between documents
    separator = tokenizer.eos_token
    full_text = separator.join(texts)

    # Tokenize entire corpus at once (no padding)
    tokenized = tokenizer(
        full_text,
        truncation=False,    # Don't truncate — we want the full corpus
        return_tensors='pt'
    )
    all_input_ids = tokenized['input_ids'][0]   # (total_tokens,)

    # Chunk into blocks of max_length
    n_tokens = len(all_input_ids)
    n_blocks = n_tokens // max_length
    blocks   = all_input_ids[:n_blocks * max_length].view(n_blocks, max_length)

    print(f"Total tokens:  {n_tokens:,}")
    print(f"Block size:    {max_length}")
    print(f"Total blocks:  {n_blocks}")

    # Convert to HuggingFace Dataset
    return Dataset.from_dict({
        "input_ids": blocks.tolist(),
        "attention_mask": torch.ones_like(blocks).tolist()
    })
```

```python
gen_tok.pad_token = gen_tok.eos_token

def generate_text(
    prompt: str,
    strategy: str = "sample",
    max_new_tokens: int = 100,
    num_sequences: int = 1
) -> list[str]:
    """
    Compare different decoding strategies:

    1. Greedy:       Always pick highest probability token
                     Fast, but repetitive
    2. Beam Search:  Explore top-k paths, return best overall
                     Higher quality, no randomness
    3. Sampling:     Sample from distribution
                     Creative, but may be incoherent
    4. Top-k:        Sample from top k tokens only
                     Balanced quality/diversity
    5. Top-p:        Sample from tokens summing to probability p
                     Better than top-k (adapts to distribution)
    6. Temperature:  Scale the distribution (low=focused, high=creative)
    """
    inputs    = gen_tok(prompt, return_tensors='pt')
    input_len = inputs['input_ids'].shape[1]

    strategies = {
        "greedy": {
            "do_sample": False, "num_beams": 1
        },
        "beam": {
            "do_sample": False, "num_beams": 5, "early_stopping": True
        },
        "sample": {
            "do_sample": True, "temperature": 0.8
        },
        "top_k": {
            "do_sample": True, "top_k": 50, "temperature": 0.8
        },
        "nucleus": {
            "do_sample": True, "top_p": 0.92, "temperature": 0.8
        },
    }

    gen_kwargs = strategies.get(strategy, strategies["nucleus"])

    with torch.no_grad():
        outputs = gen_model.generate(
            **inputs,
            max_new_tokens=max_new_tokens,
            num_return_sequences=num_sequences,
            pad_token_id=gen_tok.eos_token_id,
            repetition_penalty=1.2,   # Penalize repeating tokens
            **gen_kwargs
        )

    # Decode only the newly generated tokens
    generated = []
    for output in outputs:
        new_tokens = output[input_len:]
        generated.append(gen_tok.decode(new_tokens, skip_special_tokens=True))

    return generated


# Compare strategies
prompt = "Python decorators allow you to"
print(f"Prompt: '{prompt}'\n")

for strategy in ["greedy", "beam", "top_k", "nucleus"]:
    result = generate_text(prompt, strategy=strategy)[0]
    print(f"{strategy:10}: {result[:120]}")
    print()
```

### 📖 T5 Architecture Overview

```python
"""
T5 (Text-to-Text Transfer Transformer) — Raffel et al. (2020)

Key innovation: EVERY NLP task is framed as text-to-text generation.

  Classification: "sentiment: This movie was great."  → "positive"
  Summarization:  "summarize: Long article text..."    → "Short summary"
  Translation:    "translate English to French: Hello" → "Bonjour"
  QA:             "question: What is... context: ..."  → "The answer is..."
  SQL:            "translate to SQL: Show all users"   → "SELECT * FROM users"

Architecture:
  - Full Encoder-Decoder Transformer (unlike BERT=encoder-only, GPT=decoder-only)
  - Encoder: bidirectional self-attention (like BERT)
  - Decoder: causal self-attention + cross-attention to encoder
  - Pre-training: span corruption (mask random spans, predict them)

T5 variants:
  t5-small:  60M params
  t5-base:   220M params
  t5-large:  770M params
  t5-3b:     3B params
  t5-11b:    11B params
  flan-t5:   instruction-tuned T5 — best for few-shot
"""
```

```python
def tokenize_for_summarization(batch, prefix: str = "summarize: "):
    """
    Prepare batch for seq2seq fine-tuning.
    Inputs: prefix + article text
    Labels: summary text (tokenized separately with max_target)
    """
    # Tokenize inputs (article)
    inputs = tokenizer(
        [prefix + article for article in batch["article"]],
        truncation=True,
        max_length=MAX_INPUT,
        padding=False,
    )

    with tokenizer.as_target_tokenizer():
        targets = tokenizer(
            batch["highlights"],
            truncation=True,
            max_length=MAX_TARGET,
            padding=False,
        )

    inputs["labels"] = targets["input_ids"]
    return inputs


def tokenize_for_classification(batch, prefix: str = "sentiment analysis: "):
    """
    T5-style classification: label is a text string, not an integer.
    The model generates the label as text.
    """
    label_map = {0: "negative", 1: "positive"}

    inputs = tokenizer(
        [prefix + text for text in batch["text"]],
        truncation=True,
        max_length=MAX_INPUT,
        padding=False,
    )

    # Convert int labels to text strings
    label_texts = [label_map[l] for l in batch["label"]]
    with tokenizer.as_target_tokenizer():
        targets = tokenizer(
            label_texts,
            truncation=True,
            max_length=10,   # "negative" = 2 tokens, very short
            padding=False,

    inputs["labels"] = targets["input_ids"]
```

### 🚀 Seq2SeqTrainer with ROUGE Evaluation

```python
from transformers import (
    T5ForConditionalGeneration,
    DataCollatorForSeq2Seq,
    Seq2SeqTrainingArguments,
    Seq2SeqTrainer,
)
import evaluate

# ── Model ────────────────────────────────────────────────────────────
model = T5ForConditionalGeneration.from_pretrained(MODEL_NAME)

# ── Data Collator ─────────────────────────────────────────────────────
# Handles padding of both inputs AND labels (label padding with -100)
# -100 is ignored in the loss computation
seq2seq_collator = DataCollatorForSeq2Seq(
    tokenizer=tokenizer,
    model=model,
    padding='longest',
    label_pad_token_id=-100   # -100 ignored in CrossEntropyLoss
)

# ── ROUGE Metrics ─────────────────────────────────────────────────────
rouge = evaluate.load("rouge")

def compute_metrics(eval_preds):
    predictions, labels = eval_preds

    # Replace -100 in labels with pad token id
    labels = np.where(labels != -100, labels, tokenizer.pad_token_id)

    # Decode predictions and labels
    decoded_preds  = tokenizer.batch_decode(predictions, skip_special_tokens=True)
    decoded_labels = tokenizer.batch_decode(labels,      skip_special_tokens=True)

    # ROUGE expects newline-separated sentences
    decoded_preds  = ["\n".join(p.strip().split(". ")) for p in decoded_preds]
    decoded_labels = ["\n".join(l.strip().split(". ")) for l in decoded_labels]

    result = rouge.compute(
        predictions=decoded_preds,
        references=decoded_labels,
        use_stemmer=True
    )

    # Add average generation length
    prediction_lens = [np.count_nonzero(pred != tokenizer.pad_token_id)
                        for pred in predictions]
    result["gen_len"] = np.mean(prediction_lens)

    return {k: round(v, 4) for k, v in result.items()}


# ── Seq2Seq Training Arguments ─────────────────────────────────────────
training_args = Seq2SeqTrainingArguments(
    output_dir='./t5-small-summarization',
```

```python
def t5_inference(text: str, task: str, model, tokenizer,
                  max_new_tokens: int = 50, num_beams: int = 4) -> str:
    """
    Unified T5 inference for any task-prefixed input.
    """
    prefix_map = {
        "summarize":   "summarize: ",
        "sentiment":   "sentiment analysis: ",
        "grammar":     "grammar: ",
    }
    prefix = prefix_map.get(task, f"{task}: ")
    input_text = prefix + text

    inputs = tokenizer(input_text, return_tensors='pt',
                       truncation=True, max_length=MAX_INPUT)
    inputs = {k: v.to(model.device) for k, v in inputs.items()}

    with torch.no_grad():
        outputs = model.generate(
            **inputs,
            max_new_tokens=max_new_tokens,
            num_beams=num_beams,
            early_stopping=True,
            length_penalty=2.0   # Encourages longer sequences
        )

    return tokenizer.decode(outputs[0], skip_special_tokens=True)
```

### 🎯 PEFT with LoRA — Efficient Fine-tuning

```python
from peft import LoraConfig, get_peft_model, TaskType, PeftModel
from transformers import AutoModelForSequenceClassification, AutoTokenizer
import torch

"""
LoRA (Low-Rank Adaptation) — Hu et al. (2021)

Instead of updating the full weight matrix W ∈ R^(d×d):
  Full fine-tuning:  W_new = W_pretrained + ΔW   (d×d = millions of params)
  LoRA:              W_new = W_pretrained + A·B   where A ∈ R^(d×r), B ∈ R^(r×d)
                                                   r << d (rank, typically 4-64)

For BERT-base with r=8:
  Full fine-tuning:  110M trainable params
  LoRA (q,v only):   ~300K trainable params  ← 367× fewer!
  LoRA (all layers): ~2M   trainable params  ← 55× fewer!

Why it works:
  The "task-specific" adaptation has low intrinsic dimensionality.
  A low-rank update is sufficient to specialize the model.
"""

def finetune_with_lora(
    model_name: str = 'distilbert-base-uncased',
    task_type:  str = 'classification',
    num_classes: int = 2,
    rank: int = 8,
    alpha: int = 16,
    dropout: float = 0.05,
    target_modules: list[str] = None
):
    # Load base model (frozen by default in PEFT)
    base_model = AutoModelForSequenceClassification.from_pretrained(
        model_name,
        num_labels=num_classes
    )

    # LoRA configuration
    lora_config = LoraConfig(
        task_type=TaskType.SEQ_CLS,
        r=rank,                         # Rank of the low-rank update
        lora_alpha=alpha,               # Scaling factor: ΔW = (alpha/r) * A*B
        lora_dropout=dropout,
        target_modules=target_modules or ["q_lin", "v_lin"],  # DistilBERT
        bias="none",                    # Don't add LoRA to bias terms
        inference_mode=False
    )

    # Wrap with PEFT
    model = get_peft_model(base_model, lora_config)
    model.print_trainable_parameters()

    return model, lora_config


# Compare parameter counts
print("=== Parameter Comparison ===\n")

print(f"  Total:     {full_params:,}")
print(f"  Trainable: {trainable_full:,} (100%)")
```

### 🔬 LoRA Training Loop

```python
from transformers import TrainingArguments, Trainer
from datasets import load_dataset
import evaluate, numpy as np

def train_with_lora(lora_model, tokenizer, dataset):
    accuracy = evaluate.load("accuracy")

    def compute_metrics(eval_pred):
        logits, labels = eval_pred
        preds = np.argmax(logits, axis=-1)
        return accuracy.compute(predictions=preds, references=labels)

    def tokenize(batch):
        return tokenizer(batch['text'], truncation=True, max_length=256)

    tokenized = dataset.map(tokenize, batched=True, remove_columns=['text'])

    training_args = TrainingArguments(
        output_dir='./distilbert-lora-imdb',
        num_train_epochs=3,
        per_device_train_batch_size=32,
        per_device_eval_batch_size=64,
        learning_rate=3e-4,     # LoRA can use higher LR than full fine-tuning
        evaluation_strategy="epoch",
        save_strategy="epoch",
        load_best_model_at_end=True,
        metric_for_best_model="accuracy",
        fp16=True,
        logging_steps=50,
        report_to="none",
    )

    trainer = Trainer(
        model=lora_model,
        args=training_args,
        train_dataset=tokenized['train'],
        eval_dataset=tokenized['test'],
        tokenizer=tokenizer,
        compute_metrics=compute_metrics,
    )

    trainer.train()

    # Save only the LoRA adapters (not the full model!)
    # This is just ~2MB instead of ~250MB for the full model
    lora_model.save_pretrained('./distilbert-lora-adapters')
    print("LoRA adapters saved (only the adapter weights, not the base model)")

    return trainer.evaluate()


# Load and merge LoRA adapters
def load_lora_for_inference(base_model_name: str, adapter_path: str):
    """
    Load base model + LoRA adapters and optionally merge them.
    Merging = fold LoRA weights into base → same speed as full model, no PEFT overhead.
    """
    base   = AutoModelForSequenceClassification.from_pretrained(base_model_name, num_labels=2)
    model  = PeftModel.from_pretrained(base, adapter_path)

    # Option 1: Keep adapters separate (smaller on disk, slightly slower inference)
    # model.eval()

    # Option 2: Merge adapters into base model (faster inference, larger on disk)
    merged = model.merge_and_unload()
    merged.eval()
    print("LoRA adapters merged into base model")
    return merged
```

```python
class ModelComparison:
    """
    Systematic comparison of all fine-tuned models on the same task.
    """

    def __init__(self, test_texts: list[str], true_labels: list[int]):
        self.test_texts  = test_texts
        self.true_labels = true_labels

    def evaluate_model(self, name: str, predict_fn, model_size_mb: float = None) -> dict:
        import time
        from sklearn.metrics import accuracy_score, f1_score, classification_report

        start    = time.time()
        preds    = predict_fn(self.test_texts)
        elapsed  = time.time() - start

        acc = accuracy_score(self.true_labels, preds)
        f1  = f1_score(self.true_labels, preds, average='weighted')

        result = {
            "model":            name,
            "accuracy":         acc,
            "f1_weighted":      f1,
            "latency_ms":       (elapsed / len(self.test_texts)) * 1000,
            "total_time_s":     elapsed,
            "model_size_mb":    model_size_mb,
        }
        print(f"\n{name}:")
        print(f"  Accuracy:  {acc:.4f}")
        print(f"  F1:        {f1:.4f}")
        print(f"  Latency:   {result['latency_ms']:.1f}ms/sample")
        if model_size_mb:
            print(f"  Size:      {model_size_mb:.0f}MB")

        return result

    def print_summary_table(self, results: list[dict]):
        print("\n" + "="*70)
```

## ⚠️ Top 8 Fine-tuning Pitfalls

1. **Learning rate too high** — Use `2e-5` for BERT, `5e-5` for T5, `3e-4` for LoRA. Start with these.
2. **Forgetting `model.train()` / `model.eval()`** — Dropout and BatchNorm behave differently in each mode.
3. **Labels not shifted in CLM** — `DataCollatorForLanguageModeling` does this; don't do it manually.
4. **Not using `label=-100` for padding** — Padding tokens in labels will contribute to loss incorrectly.
5. **Tokenizer mismatch** — Always save and load the SAME tokenizer as the model. Never mix.
6. **Evaluating with generate() but training with teacher forcing** — Train-eval mismatch in T5. Use `predict_with_generate=True`.
7. **No gradient clipping** — Add `max_grad_norm=1.0` to `TrainingArguments`. Essential for stability.
8. **Overfitting on small datasets** — Use LoRA + strong dropout + small learning rate for <1K examples.
